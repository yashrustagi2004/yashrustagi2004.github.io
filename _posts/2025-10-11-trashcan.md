What I understood

1. The Recycle Bin is located in the root of the OS drive under $Recycle.Bin. 

2. When a file is deleted, two files are created there:

    $I file: contains metadata (original path, size, deletion time)

    $R file: contains the actual deleted data

3. Both files share the same 6-character suffix:
    Example: $RSJTV6E.png and $ISJTV6E.png


By glancing at the files, I noticed that there are more $I files than $R files so that was something weird. I proceeded to get the extact count

```bash
~/Downloads/trashcan ❯ ls -l \$I* | wc -l
117
~/Downloads/trashcan ❯ ls -l \$R* | wc -l
78
```

Okay and 117-78 is 39, which is the flag length! (38 actually) 
So that means my next step is to filter out these unique $I files for which their corressponding $R file is not there

## 2. Filtering out $I files

For this, I used the below code

```python
#!/usr/bin/env python3
import glob
import os
import shutil

# Set paths
source_dir = os.path.expanduser('~/Downloads/trashcan')
output_dir = os.path.expanduser('~/Downloads/trashcan/orphaned_I_files')

# Create output directory if it doesn't exist
os.makedirs(output_dir, exist_ok=True)

# Get all $I and $R files
I_files = glob.glob(os.path.join(source_dir, '$I*.txt'))
R_files = glob.glob(os.path.join(source_dir, '$R*.txt'))

print(f"Found {len(I_files)} $I files")
print(f"Found {len(R_files)} $R files")

# Extract the identifiers (the part after $I and $R)
# For example: $IL4WLXW.txt -> L4WLXW
I_identifiers = {}
for f in I_files:
    basename = os.path.basename(f)
    # Remove $I prefix and .txt suffix
    identifier = basename[2:-4]  # $I + identifier + .txt
    I_identifiers[identifier] = f

R_identifiers = set()
for f in R_files:
    basename = os.path.basename(f)
    # Remove $R prefix and .txt suffix
    identifier = basename[2:-4]  # $R + identifier + .txt
    R_identifiers.add(identifier)

# Find orphaned $I files (those without corresponding $R files)
orphaned_I_files = []
for identifier, filepath in I_identifiers.items():
    if identifier not in R_identifiers:
        orphaned_I_files.append(filepath)

print(f"\nFound {len(orphaned_I_files)} orphaned $I files (no matching $R file)")
print("="*80)

# Copy orphaned files to output directory
for filepath in orphaned_I_files:
    basename = os.path.basename(filepath)
    dest_path = os.path.join(output_dir, basename)
    shutil.copy2(filepath, dest_path)
    print(f"Copied: {basename}")

print("="*80)
print(f"\nOrphaned $I files copied to: {output_dir}")
print(f"Total orphaned files: {len(orphaned_I_files)}")

# Show summary
if len(orphaned_I_files) == 39:
    print("\n✓ Exactly 39 orphaned files found (as expected)!")
else:
    print(f"\n⚠ Expected 39 orphaned files, but found {len(orphaned_I_files)}")
```

## 3. Retrieving the flag


```python
#!/usr/bin/env python3
import os
import struct

# Directory containing $I files
idir = '.'

# Helper: get FILETIME from $I file (little-endian 8 bytes at offset 16)
def get_filetime(path):
    try:
        with open(path, 'rb') as f:
            f.seek(16)
            data = f.read(8)
            if len(data) == 8:
                return struct.unpack('<Q', data)[0]
    except Exception:
        pass
    return 0

# Helper: extract 9th byte and convert to printable char if possible
def extract_9th_byte(path):
    try:
        b = open(path, 'rb').read()
        if len(b) > 8:
            val = b[8]
            ch = chr(val) if 32 <= val <= 126 else '.'
            return val, ch
    except Exception:
        pass
    return None, '.'

# Collect all $I files
files = [f for f in os.listdir(idir) if f.startswith('$I') and os.path.isfile(f)]
# Sort ascending by FILETIME
files.sort(key=get_filetime)

stream = []

# Extract 9th byte from each file
for f in files:
    val, ch = extract_9th_byte(f)
    stream.append(ch)
    print(f"{f:25}  9th byte: {val} ('{ch}')")

# Join all characters into a string
result = ''.join(stream)
print("\n[+] Reconstructed 9th-byte stream:")
print(result)

# Save to file
with open('extracted_9th_chars.txt', 'w') as fo:
    fo.write(result)

print("\n[+] Saved to extracted_9th_chars.txt")
```

The script is first sorting the files in ascending order based on time of deleting and then extracting the 9th byte from each flag
